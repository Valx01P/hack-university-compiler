#!/usr/bin/env bash
# A sample for the tokenizer snapshots: the rules the Bash course needs.
# shellcheck disable=SC2034
set -euo pipefail

name="World"
greeting='single #quoted'
count=${#name}
first=${name:0:1}
path=/srv/app/releases/v1.2.3
base=${path##*/}
dir=${path%/*}
ext=${file#*.}
args=$#
echo "Hello, $name! You have $# args and ${#greeting} chars."
echo 'It'\''s a #hash' $'tab\there\nnewline \x41 \u00e9 \e[0m \q' $"locale"
echo "escapes: \$HOME \`cmd\` \"quoted\" \\ and \n stays" # a real comment
echo a#b c#d url=https://example.com/#section #comment after url
echo \#escaped not-a-comment
arr=(one two "three four") # an array
echo "${arr[@]}" "${#arr[@]}" "$((1 + 2))" "$(date +%F)"
total=$(( count * 16#ff + 0x10 ))
(( i = 2#101 ))
for ((i = 0; i < 3; i++)); do
  echo "$i"
done
for f in *.txt; do
  [[ -f "$f" ]] && wc -l "$f" # count lines
done
while read -r line; do echo "$line"; done < input.txt
case "$1" in
  start|run) echo "starting";;
  stop) echo "stopping" ;;
  *) echo "usage: $0 {start|stop}" >&2; exit 1 ;;
esac
greet() {
  local who="${1:-friend}"
  printf 'Hello, %s\n' "$who"
  return 0
}
function cleanup {
  rm -f "$tmp" 2>/dev/null || true
}
trap cleanup EXIT
tmp=$(mktemp)
result="$(basename "$path")"
nested="$(echo "inner $(echo deep)")"
files=$(
  # every log, even the hidden ones
  find . -name '*.log'
)
cat <<EOF
# not a comment: part of the heredoc
Home is $HOME and \$HOME, cost \`$5\`
EOF
cat <<-'RAW'
	$HOME stays \n raw
	RAW
cat <<< "here string" # comment
npm test && docker exec app ls
echo done in time # keywords as arguments
if [ "$count" -gt 3 ]; then echo big; elif [ "$count" -eq 0 ]; then echo none; else echo small; fi
# TODO: print the totals
#       one per line
exit 0
