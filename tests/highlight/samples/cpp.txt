#include <iostream>
#include <vector>
#include <string>
using namespace std;

template <typename T>
T biggest(const vector<T>& xs) {
    T best = xs[0];
    for (const auto& x : xs) {
        if (x > best) best = x;  // keep the largest
    }
    return best;
}
