import { useState, type ReactNode } from "react";

type Props = { title: string; children?: ReactNode };

export function Card({ title, children }: Props) {
  const [open, setOpen] = useState<boolean>(false);
  return (
    <section className="card" data-open={open}>
      {/* the header toggles the body */}
      <h2 onClick={() => setOpen(!open)}>{title}</h2>
      {open && <div className='body'>{children}</div>}
      <p>it's {open ? "open" : "closed"} now</p>
      {
        /* a comment on
           two lines */
      }
      <Button kind="primary" onPress={() => alert(`Hi ${title}\n`)} />
      <>
        <input disabled value={title} />
      </>
    </section>
  );
}

function Button({ kind, onPress }: { kind: string; onPress(): void }) {
  return <button className={`btn btn-${kind}`} onClick={onPress}>Go</button>;
}
