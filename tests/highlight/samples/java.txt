import java.util.*;
import java.util.function.Function;

/**
 * Javadoc with {@link List} and {@code x < y}.
 */
@FunctionalInterface
interface Shape {
    double area();
}

public class Main {
    @Override
    public String toString() {
        return "Main";
    }

    @SuppressWarnings("unchecked")
    static <T> List<T> listOf(T... items) {
        return new ArrayList<>(Arrays.asList(items));
    }

    public @interface Marker {}

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = "tab\tquote\" unicode\u0041 octal\101";
        char c = '\n';
        int n = 1_000;
        long big = 42L;
        double d = 2.5e3;
        var list = listOf(1, 2, 3);
        Function<Integer, Integer> square = x -> x * x;
        System.out.println(s + c + n + big + d + list + square.apply(4)); // prints all
        // TODO: read a line with in.nextLine()
        //       and print it back
    }
}
